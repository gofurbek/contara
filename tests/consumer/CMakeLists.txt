cmake_minimum_required(VERSION 3.21)
project(contara_consumer LANGUAGES CXX)
find_package(contara 0.1 CONFIG REQUIRED)
add_executable(consumer main.cpp)
target_link_libraries(consumer PRIVATE contara::contara)
if(MSVC)
    target_compile_options(consumer PRIVATE /EHsc)
endif()
enable_testing()
add_test(NAME installed_package COMMAND consumer)
